import { tc } from "../../../../shared/styles/theme-color";
import { TransformControls } from "@react-three/drei";
import { useEffect, useRef } from "react";
import * as THREE from "three";
import type { TheaterLayout } from "../../../../shared/types/script";

const snapValue = (value: number, step: number, enabled: boolean) => {
  if (!enabled || step <= 0) return value;
  return Math.round(value / step) * step;
};

export const AudienceSeatsHandle = ({
  layout,
  active,
  snapEnabled,
  snapStep,
  onAudienceStartZChange,
  onDraggingChange,
}: {
  layout: TheaterLayout;
  active: boolean;
  snapEnabled: boolean;
  snapStep: number;
  onAudienceStartZChange: (nextZ: number) => void;
  onDraggingChange: (value: boolean) => void;
}) => {
  const groupRef = useRef<THREE.Group>(null);
  const halfDepth = layout.hallDepth / 2;
  const minZ = -halfDepth + 0.4;
  const maxZ = halfDepth - 0.4;
  const blockDepth =
    layout.seatRows > 0
      ? (layout.seatRows - 1) * layout.rowSpacing
      : 0;
  const centerZ = layout.audienceStartZ + blockDepth / 2;
  const width = Math.max(
    1,
    (layout.seatsPerRow - 1) * layout.seatSpacing + layout.aisleWidth,
  );

  useEffect(() => {
    if (!groupRef.current) return;
    groupRef.current.position.set(0, 0.45, THREE.MathUtils.clamp(centerZ, minZ, maxZ));
  }, [centerZ, minZ, maxZ]);

  const commitPosition = () => {
    if (!groupRef.current) return;
    const draggedCenterZ = groupRef.current.position.z;
    const nextStartZ = THREE.MathUtils.clamp(
      snapValue(draggedCenterZ - blockDepth / 2, snapStep, snapEnabled),
      minZ,
      maxZ - blockDepth,
    );
    onAudienceStartZChange(nextStartZ);
    groupRef.current.position.z = nextStartZ + blockDepth / 2;
  };

  if (!active || layout.seatRows <= 0) return null;

  return (
    <>
      <group ref={groupRef}>
        <mesh>
          <boxGeometry args={[width, 0.12, Math.max(0.35, blockDepth + 0.35)]} />
          <meshStandardMaterial
            color={tc("--color-active-ascent")}
            transparent
            opacity={0.22}
            depthWrite={false}
          />
        </mesh>
      </group>
      {groupRef.current?.parent && (
        <TransformControls
          object={groupRef.current}
          mode="translate"
          showX={false}
          showY={false}
          showZ
          onMouseDown={() => onDraggingChange(true)}
          onMouseUp={() => {
            onDraggingChange(false);
            commitPosition();
          }}
          onObjectChange={commitPosition}
        />
      )}
    </>
  );
};
